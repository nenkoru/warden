+package relay

+import "runtime/debug"
+
+const quicPath = "github.com/quic-go/quic-go"
+
+// Version reports the version of quic-go built into the binary, or "unknown".
+func Version() string {
+	bi, ok := debug.ReadBuildInfo()
+	if !ok {
+		return "unknown"
+	}
+	return quicVersion(bi)
+}
+
+func quicVersion(bi *debug.BuildInfo) string {
+	for _, m := range bi.Deps {
+		if m.Path != quicPath {
+			continue
+		}
+		if m.Replace != nil && m.Replace.Version != "" {
+			return m.Replace.Version
+		}
+		if m.Version != "" {
+			return m.Version
+		}
+	}
+	return "unknown"
+}
